+// M-IRTAM-F2 (P3): gate lgdc compartido (ver LgdcPacing.h).
+#include "Data/LgdcPacing.h"
+
+namespace lgdcpacing {
+namespace {
+
+// Choke point unico de la app: ultimo lanzamiento lgdc en ms (0 = ninguno).
+// Function-local static (Meyers): una sola instancia en todo el proceso,
+// compartida por catalogo + getbest + gambit.
+std::atomic<long long>& gate() {
+    static std::atomic<long long> last{0};
+    return last;
+}
+
+}  // namespace
+
+void recordLaunch(long long nowMs) {
+    // CAS-max: solo avanzar. Un worker tardio no retrasa el reloj.
+    long long cur = gate().load(std::memory_order_relaxed);
+    while (nowMs > cur) {
+        if (gate().compare_exchange_weak(cur, nowMs,
+                                         std::memory_order_relaxed)) {
+            break;
+        }
+    }
+}
+
+long long lastLaunchMs() {
+    return gate().load(std::memory_order_relaxed);
+}
+
+long long msSinceLastLaunch(long long nowMs) {
+    return nowMs - lastLaunchMs();
+}
+
+bool canLaunchGambit(long long nowMs) {
+    return msSinceLastLaunch(nowMs) >= kGambitGapMs;
+}
+
+void resetForTest() {
+    gate().store(0, std::memory_order_relaxed);
+}
+
+}  // namespace lgdcpacing

+#pragma once
+// =============================================================================
+// M-IRTAM-F2 (P3): gate lgdc compartido (C1) — choke point unico de pacing.
+//
+// Modulo PURO: solo std (atomic/cstdint). Sin curl, sin GL, sin App, sin URLs,
+// sin hosts, sin endpoints. Testeable en CI (tests/test_irtamc_gate.cpp) con
+// reloj simulado: toda la API recibe nowMs explicito (ms desde Unix epoch,
+// misma base que el msNow de GiroAdapter::pollRound).
+//
+// Invariante C1: gambit NUNCA a <15 s de CUALQUIER request lgdc de la app
+// (catalogo incluido). El pacing M4 de getbest queda intacto y grandfathered:
+// GiroAdapter solo REGISTRA sus lanzamientos (catalogo + punto CAS de getbest),
+// cero cambio de comportamiento. El fetcher gambit (P4) CONSULTA + REGISTRA.
+//
+// Semantica forward-only: recordLaunch solo avanza el reloj (CAS-max). Dos
+// workers getbest registrando fuera de orden no pueden retrasar el reloj y
+// abrir una ventana gambit prematura.
+// =============================================================================
+#include <atomic>
+#include <cstdint>
+
+namespace lgdcpacing {
+
+// Nombres declarados para el fold (grep): lgdcpacing::recordLaunch,
+// lgdcpacing::canLaunchGambit, lgdcpacing::lastLaunchMs,
+// lgdcpacing::msSinceLastLaunch, lgdcpacing::kGambitGapMs.
+
+inline constexpr long long kGambitGapMs = 15000;  // 1 req/15 s POR CLIENTE (E11)
+
+// Registra un lanzamiento lgdc (catalogo, getbest o gambit) en nowMs.
+// Solo avanza: un registro tardio con timestamp viejo no mueve el reloj.
+void recordLaunch(long long nowMs);
+
+// Ultimo lanzamiento registrado (0 = ninguno: arranque frio).
+long long lastLaunchMs();
+
+// nowMs - lastLaunchMs() (puede ser negativo si el reloj simulado retrocede).
+long long msSinceLastLaunch(long long nowMs);
+
+// true si gambit puede lanzarse en nowMs: >= kGambitGapMs desde CUALQUIER
+// lanzamiento lgdc previo. Arranque frio (sin registros) = true.
+bool canLaunchGambit(long long nowMs);
+
+// Solo tests: vuelve a arranque frio. Patron FASE B (launchGapSec publica
+// para tests): el hook de test vive en la cabecera de produccion.
+void resetForTest();
+
+}  // namespace lgdcpacing

+// =============================================================================
+// test_irtamc_gate — gate lgdc compartido (M-IRTAM-F2 P3, C1, spec-hija).
+//
+// Test de logica pura: sin GL, sin red, reloj SIMULADO (nowMs explicito).
+// Llama a PRODUCCION (namespace lgdcpacing en src/Data/LgdcPacing.{h,cpp}),
+// no replica. Cubre F2 G8: primitiva unica, gap >= 15 s desde CUALQUIER
+// lanzamiento previo, secuencia catalogo->getbest->gambit, rechazo en burst,
+// semantica forward-only, arranque frio.
+//
+// El pacing M4 de getbest NO se toca aqui (regresion cubierta por los tests
+// existentes + ctest global): este test prueba el gate, no el adapter.
+// =============================================================================
+#include "Data/LgdcPacing.h"
+
+#include <cstdio>
+
+static int g_ok = 0, g_fail = 0;
+
+static void check_bool(const char* what, bool cond) {
+    if (cond) ++g_ok;
+    else ++g_fail;
+    std::printf("[%s] %s\n", cond ? "OK" : "FAIL", what);
+}
+
+int main() {
+    using lgdcpacing::canLaunchGambit;
+    using lgdcpacing::kGambitGapMs;
+    using lgdcpacing::lastLaunchMs;
+    using lgdcpacing::msSinceLastLaunch;
+    using lgdcpacing::recordLaunch;
+    using lgdcpacing::resetForTest;
+
+    std::printf("--- F2 P3: arranque frio + borde 15 s ---\n");
+    resetForTest();
+    check_bool("gap declarado 15000 ms", kGambitGapMs == 15000);
+    check_bool("frio: last == 0", lastLaunchMs() == 0);
+    check_bool("frio: gambit permitido", canLaunchGambit(1000000));
+
+    resetForTest();
+    recordLaunch(1000000);
+    check_bool("registra last", lastLaunchMs() == 1000000);
+    check_bool("gap 0 recien registrado", msSinceLastLaunch(1000000) == 0);
+    check_bool("14999 ms: denegado", !canLaunchGambit(1000000 + 14999));
+    check_bool("15000 ms: permitido (borde >=)", canLaunchGambit(1000000 + 15000));
+    check_bool("15001 ms: permitido", canLaunchGambit(1000000 + 15001));
+
+    std::printf("--- F2 P3: secuencia catalogo->getbest->gambit ---\n");
+    resetForTest();
+    recordLaunch(2000000);          // catalogo (fetchCatalog instrumentado)
+    check_bool("tras catalogo: gambit denegado a +1 s",
+               !canLaunchGambit(2000000 + 1000));
+    recordLaunch(2000000 + 5000);   // getbest (punto CAS instrumentado)
+    check_bool("last = getbest (el mas reciente)", lastLaunchMs() == 2005000);
+    // El gap se mide desde CUALQUIER previo: el mas reciente manda.
+    check_bool("gambit a +14 s del catalogo pero +9 s del getbest: denegado",
+               !canLaunchGambit(2000000 + 14000));
+    check_bool("gambit a +15 s del getbest: permitido",
+               canLaunchGambit(2005000 + 15000));
+    check_bool("gambit a +15 s - 1 ms del getbest: denegado",
+               !canLaunchGambit(2005000 + 14999));
+
+    std::printf("--- F2 P3: burst + forward-only ---\n");
+    resetForTest();
+    recordLaunch(3000000);
+    check_bool("burst a +10 ms: denegado", !canLaunchGambit(3000010));
+    check_bool("burst a +1000 ms: denegado", !canLaunchGambit(3001000));
+    recordLaunch(3000000 - 5000);   // registro tardio (worker fuera de orden)
+    check_bool("stale no retrasa el reloj", lastLaunchMs() == 3000000);
+    check_bool("gap sigue midiendose desde el maximo",
+               msSinceLastLaunch(3015000) == 15000);
+    check_bool("tras ventana: permitido", canLaunchGambit(3015000));
+    resetForTest();
+    check_bool("reset aisla casos", lastLaunchMs() == 0);
+
+    std::printf("\n[test_irtamc_gate] %d OK, %d FAIL\n", g_ok, g_fail);
+    return g_fail == 0 ? 0 : 1;
+}
