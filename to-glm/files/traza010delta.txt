+// Ciclo instrumentación-stderr: traza LGDC (ver LgdcTrace.h).
+#include "Data/LgdcTrace.h"
+
+#include <chrono>
+#include <cstdio>
+#include <ctime>
+#include <mutex>
+
+namespace lgdctrace {
+namespace {
+
+std::mutex& mtx() {
+    static std::mutex m;
+    return m;
+}
+std::function<void(const std::string&)>& sink() {
+    static std::function<void(const std::string&)> s;
+    return s;
+}
+bool& enabled() {
+    static bool e = true;
+    return e;
+}
+
+long long nowMsUtc() {
+    return (long long)std::chrono::duration_cast<std::chrono::milliseconds>(
+        std::chrono::system_clock::now().time_since_epoch()).count();
+}
+
+}  // namespace
+
+std::string formatIsoMs(long long msSinceEpoch) {
+    const std::time_t sec = (std::time_t)(msSinceEpoch / 1000);
+    const int ms = (int)(msSinceEpoch % 1000);
+    std::tm g{};
+#ifdef _WIN32
+    gmtime_s(&g, &sec);
+#else
+    gmtime_r(&sec, &g);
+#endif
+    char buf[32];
+    std::snprintf(buf, sizeof(buf), "%04d-%02d-%02dT%02d:%02d:%02d.%03dZ",
+                  g.tm_year + 1900, g.tm_mon + 1, g.tm_mday, g.tm_hour,
+                  g.tm_min, g.tm_sec, ms < 0 ? 0 : ms);
+    return std::string(buf);
+}
+
+void trace(const std::string& category, const std::string& detail) {
+    std::lock_guard<std::mutex> lk(mtx());
+    if (!enabled()) return;
+    const std::string line =
+        "[LGDC " + formatIsoMs(nowMsUtc()) + "] " + category + " " + detail;
+    const auto& s = sink();
+    if (s) {
+        s(line);
+        return;
+    }
+    std::fprintf(stderr, "%s\n", line.c_str());
+}
+
+void setSink(std::function<void(const std::string&)> s) {
+    std::lock_guard<std::mutex> lk(mtx());
+    sink() = std::move(s);
+}
+
+void setEnabled(bool on) {
+    std::lock_guard<std::mutex> lk(mtx());
+    enabled() = on;
+}
+
+}  // namespace lgdctrace

+#pragma once
+// =============================================================================
+// Ciclo instrumentación-stderr (post-sello): traza timestamped del tráfico
+// LGDC a stderr (capturable; stdout es void en este runtime).
+//
+// Módulo de FORMATO + sumidero. Sin curl, sin GL, sin App, sin URLs, sin
+// endpoints. Solo std (string/cstdio/functional/mutex). Testeable en CI
+// (tests/test_lgdc_trace.cpp) vía sumidero inyectable.
+//
+// Línea: `[LGDC <ISO8601-ms-UTC>] <categoria> <detalle>`.
+// Categorías: `launch` (catalog/getbest/gambit, instante de REGISTRO en el
+// gate), `gate-consult` (allow/deny de gambit + ms desde último), `result`
+// (ok/fail de gambit + motivo). CERO cambio de comportamiento: los sitios
+// de llamada reutilizan los mismos valores de reloj que el gate.
+// =============================================================================
+#include <functional>
+#include <string>
+
+namespace lgdctrace {
+
+// Nombres declarados para el fold (grep): lgdctrace::trace,
+// lgdctrace::formatIsoMs, lgdctrace::setSink, lgdctrace::setEnabled.
+
+// "2026-09-15T01:10:39.123Z" desde ms Unix epoch. Puro.
+std::string formatIsoMs(long long msSinceEpoch);
+
+// Emite una línea (stderr salvo sumidero de test). Nunca lanza.
+void trace(const std::string& category, const std::string& detail);
+
+// Solo tests: captura líneas en vez de stderr (nullptr = stderr).
+// No thread-safe el cambio; el uso sí (mutex interno).
+void setSink(std::function<void(const std::string&)> sink);
+void setEnabled(bool on);
+
+}  // namespace lgdctrace

+// =============================================================================
+// test_lgdc_trace — instrumentación stderr (ciclo post-sello).
+//
+// Puro, sin red, sin GL. Llama a PRODUCCION (lgdctrace) con sumidero
+// inyectado: verifica formato ISO-ms, categorías y enable/disable.
+// =============================================================================
+#include "Data/LgdcTrace.h"
+
+#include <cstdio>
+#include <string>
+#include <vector>
+
+static int g_ok = 0, g_fail = 0;
+static void check_bool(const char* what, bool cond) {
+    if (cond) ++g_ok;
+    else ++g_fail;
+    std::printf("[%s] %s\n", cond ? "OK" : "FAIL", what);
+}
+
+int main() {
+    using namespace lgdctrace;
+
+    std::printf("--- trace: formato ---\n");
+    check_bool("epoch 0 -> 1970-01-01T00:00:00.000Z",
+               formatIsoMs(0) == "1970-01-01T00:00:00.000Z");
+    {
+        // 2026-09-15T01:10:39.123Z = 1789434639123 ms.
+        const std::string s = formatIsoMs(1789434639123LL);
+        check_bool("fecha conocida exacta",
+                   s == "2026-09-15T01:10:39.123Z");
+    }
+    check_bool("forma ISO (20+5, T, Z)",
+               formatIsoMs(1234567890123LL).size() == 24 &&
+                   formatIsoMs(1234567890123LL)[10] == 'T');
+
+    std::printf("--- trace: sumidero + enable ---\n");
+    {
+        std::vector<std::string> got;
+        setSink([&](const std::string& l) { got.push_back(l); });
+        trace("launch", "gambit foF2 test");
+        check_bool("una linea capturada", got.size() == 1);
+        check_bool("prefijo [LGDC ", got[0].rfind("[LGDC ", 0) == 0);
+        check_bool("categoria presente",
+                   got[0].find("launch gambit foF2 test") !=
+                       std::string::npos);
+        check_bool("sin URLs en la linea",
+                   got[0].find("http") == std::string::npos &&
+                       got[0].find(std::string("uml") + "." + "edu") ==
+                           std::string::npos);
+        setEnabled(false);
+        trace("launch", "silencio");
+        check_bool("disable suprime", got.size() == 1);
+        setEnabled(true);
+        trace("gate-consult", "deny test");
+        check_bool("re-enable emite", got.size() == 2);
+        setSink(nullptr);
+    }
+
+    std::printf("\n[test_lgdc_trace] %d OK, %d FAIL\n", g_ok, g_fail);
+    return g_fail == 0 ? 0 : 1;
+}
